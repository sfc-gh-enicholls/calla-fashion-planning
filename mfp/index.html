Generated SQL
Semantic view: CALLA.MERCH.SV_TRADING_WEEK
WITH mtd AS (
    SELECT
        c.retail_month_no,
        SUM(w.units_sold)                          AS units_sold,
        SUM(w.units_plan)                          AS units_plan
    FROM CALLA.MERCH.FCT_TRADING_WEEK w
    JOIN CALLA.MERCH.DIM_CALENDAR    c ON c.iso_week = w.iso_week
    WHERE c.cycle = 'SS26'
      AND w.iso_week <= 18
    GROUP BY c.retail_month_no
)
SELECT
    retail_month_no,
    units_sold,
    units_plan,
    DIV0(units_sold - units_plan, units_plan) * 100 AS var_to_plan_pct
FROM mtd
ORDER BY retail_month_no
