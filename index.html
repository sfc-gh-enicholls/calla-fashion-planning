Generated SQL
Semantic view: CALLA.MERCH.SV_TRADING_WEEK
WITH style_week AS (
  SELECT
    s.style_code,
    s.style_name,
    s.denim_silhouette,
    s.retail_price,
    s.cost_price,
    w.iso_week,
    SUM(w.units_sold) AS units_sold,
    SUM(w.units_plan) AS units_plan,
    SUM(w.opening_stock) AS opening_stock,
    SUM(w.intake) AS intake,
    SUM(w.closing_stock) AS closing_stock
  FROM CALLA.MERCH.FCT_TRADING_WEEK AS w
  JOIN CALLA.MERCH.DIM_STYLE AS s USING (style_code)
  WHERE w.cycle = 'SS26'
    AND s.style_code IN ('DN-FLR-002', 'DN-SKN-014')
    AND w.iso_week <= 18
  GROUP BY 1, 2, 3, 4, 5, 6
),
recent_rate AS (
  SELECT
    style_code,
    AVG(CASE WHEN iso_week BETWEEN 15 AND 18 THEN units_sold END) AS recent_weekly_units,
    SUM(CASE WHEN iso_week <= 18 THEN units_sold END) AS sold_to_date,
    SUM(CASE WHEN iso_week <= 18 THEN units_plan END) AS plan_to_date,
    MAX(CASE WHEN iso_week = 18 THEN closing_stock END) AS current_closing_stock
  FROM style_week
  GROUP BY 1
),
markdown_scenarios AS (
  SELECT
    style_code,
    retail_price,
    cost_price,
    current_closing_stock,
    recent_weekly_units,
    GREATEST(0, 26 - 18) AS weeks_left,
    current_closing_stock - LEAST(current_closing_stock, recent_weekly_units * GREATEST(0, 26 - 18)) AS hold_terminal_units,
    current_closing_stock - LEAST(current_closing_stock, recent_weekly_units * (1 + 0.045 * 30 + 0.0009 * 30 * 30) * 4)
      - LEAST(
          GREATEST(0, current_closing_stock - LEAST(current_closing_stock, recent_weekly_units * (1 + 0.045 * 30 + 0.0009 * 30 * 30) * 4)),
          recent_weekly_units * (1 + 0.045 * 30 + 0.0009 * 30 * 30) * 0.55 * GREATEST(0, 26 - (18 + 1) - 4)
        ) AS early_mid_terminal_units,
    current_closing_stock - LEAST(current_closing_stock, recent_weekly_units * GREATEST(0, 24 - 18 - 1))
      - LEAST(
          GREATEST(0, current_closing_stock - LEAST(current_closing_stock, recent_weekly_units * GREATEST(0, 24 - 18 - 1))),
          recent_weekly_units * (1 + 0.045 * 50 + 0.0009 * 50 * 50) * 3
        )
      - LEAST(
          GREATEST(
            0,
            current_closing_stock
              - LEAST(current_closing_stock, recent_weekly_units * GREATEST(0, 24 - 18 - 1))
              - LEAST(
                  GREATEST(0, current_closing_stock - LEAST(current_closing_stock, recent_weekly_units * GREATEST(0, 24 - 18 - 1))),
                  recent_weekly_units * (1 + 0.045 * 50 + 0.0009 * 50 * 50) * 3
                )
          ),
          recent_weekly_units * (1 + 0.045 * 50 + 0.0009 * 50 * 50) * 0.55 * GREATEST(0, 26 - 24)
        ) AS late_deep_terminal_units
  FROM recent_rate
  JOIN CALLA.MERCH.DIM_STYLE USING (style_code)
  WHERE style_code = 'DN-SKN-014'
)
SELECT
  style_code,
  style_name,
  denim_silhouette,
  CASE
    WHEN style_code = 'DN-FLR-002' THEN ROUND(recent_weekly_units * 4 * (retail_price - cost_price), 0)
    WHEN style_code = 'DN-SKN-014' THEN ROUND((hold_terminal_units - late_deep_terminal_units) * retail_price * 0.72 - (hold_terminal_units - early_mid_terminal_units) * retail_price * 0.72, 0)
  END AS decision_value_gbp,
  ROUND((sold_to_date / NULLIF(plan_to_date, 0) - 1) * 100, 1) AS variance_pct,
  current_closing_stock
FROM recent_rate
JOIN CALLA.MERCH.DIM_STYLE USING (style_code)
LEFT JOIN markdown_scenarios USING (style_code)
WHERE style_code IN ('DN-FLR-002', 'DN-SKN-014')
ORDER BY decision_value_gbp DESC;
