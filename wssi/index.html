Generated SQL
Semantic view: CALLA.MERCH.SV_TRADING_WEEK
SELECT
    w.iso_week,
    s.silhouette,
    SUM(w.units_sold)                             AS units_sold,
    SUM(w.units_plan)                             AS units_plan,
    SUM(w.units_sold) / NULLIF(SUM(w.opening_stock + w.intake), 0) * 100
                                                  AS sell_through_pct
FROM CALLA.MERCH.FCT_TRADING_WEEK  AS w
JOIN CALLA.MERCH.DIM_STYLE         AS s USING (style_code)
WHERE s.silhouette IN ('skinny', 'flare')
  AND w.iso_week BETWEEN 15 AND 18
  AND w.cycle = 'SS26'
GROUP BY w.iso_week, s.silhouette
ORDER BY w.iso_week;
