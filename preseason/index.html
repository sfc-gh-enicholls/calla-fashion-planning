Generated SQL
Semantic view: SV_ASSORTMENT
WITH recent_denim_mix AS (
  SELECT
    s.denim_silhouette,
    SUM(w.units_sold) AS units_sold,
    SUM(w.units_plan) AS units_plan
  FROM CALLA.MERCH.FCT_TRADING_WEEK AS w
  JOIN CALLA.MERCH.DIM_STYLE AS s USING (style_code)
  WHERE w.cycle = 'SS26'
    AND w.iso_week BETWEEN 15 AND 18
    AND s.merch_class = 'Denim'
  GROUP BY s.denim_silhouette
), option_counts AS (
  SELECT
    s.denim_silhouette,
    SUM(CASE WHEN w.cycle = 'SS26' THEN 1 ELSE 0 END) AS option_rows
  FROM CALLA.MERCH.FCT_TRADING_WEEK AS w
  JOIN CALLA.MERCH.DIM_STYLE AS s USING (style_code)
  WHERE w.cycle = 'SS26'
    AND w.iso_week = 18
    AND s.merch_class = 'Denim'
  GROUP BY s.denim_silhouette
)
SELECT
  m.denim_silhouette,
  m.units_sold,
  m.units_plan,
  o.option_rows
FROM recent_denim_mix AS m
LEFT JOIN option_counts AS o USING (denim_silhouette)
ORDER BY m.units_sold DESC;
