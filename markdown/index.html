Generated SQL
Semantic view: CALLA.MERCH.SV_MARKDOWN
WITH current_position AS (
  SELECT
    s.style_code,
    s.style_name,
    s.retail_price,
    s.cost_price,
    SUM(CASE WHEN w.iso_week = 18 THEN w.closing_stock ELSE 0 END) AS current_stock_units,
    AVG(CASE WHEN w.iso_week BETWEEN 15 AND 18 THEN w.units_sold END) AS base_ros
  FROM CALLA.MERCH.FCT_TRADING_WEEK AS w
  JOIN CALLA.MERCH.DIM_STYLE AS s USING (style_code)
  WHERE w.cycle = 'SS26'
    AND s.style_code = 'DN-SKN-014'
  GROUP BY s.style_code, s.style_name, s.retail_price, s.cost_price
),
option_inputs AS (
  SELECT * FROM VALUES
    ('hold', 'Hold — no markdown', 0, 19, 8),
    ('early-shallow', 'Early & shallow — 20% from W20', 20, 20, 4),
    ('early-mid', 'Early & mid — 30% from W20', 30, 20, 4),
    ('late-deep', 'Late & deep — 50% from W24', 50, 24, 3)
    AS option_inputs(option_id, option_label, depth_pct, start_week, duration_weeks)
),
scored AS (
  SELECT
    option_id,
    option_label,
    depth_pct,
    start_week,
    LEAST(duration_weeks, 26 - start_week + 1) AS duration_weeks,
    current_stock_units,
    base_ros,
    retail_price,
    cost_price,
    GREATEST(0, start_week - 18 - 1) AS pre_weeks,
    GREATEST(0, 8 - GREATEST(0, start_week - 18 - 1) - LEAST(duration_weeks, 26 - start_week + 1)) AS post_weeks,
    1 + 0.045 * depth_pct + 0.0009 * depth_pct * depth_pct AS uplift
  FROM current_position
  CROSS JOIN option_inputs
),
volumes AS (
  SELECT
    *,
    LEAST(current_stock_units, base_ros * pre_weeks) AS sold_pre,
    LEAST(
      current_stock_units - LEAST(current_stock_units, base_ros * pre_weeks),
      base_ros * uplift * duration_weeks
    ) AS sold_promo,
    LEAST(
      current_stock_units - LEAST(current_stock_units, base_ros * pre_weeks) - LEAST(
        current_stock_units - LEAST(current_stock_units, base_ros * pre_weeks),
        base_ros * uplift * duration_weeks
      ),
      base_ros * uplift * 0.55 * post_weeks
    ) AS sold_post
  FROM scored
)
SELECT
  option_label,
  depth_pct,
  start_week,
  duration_weeks,
  ROUND(sold_pre + sold_promo + sold_post) AS units_cleared,
  ROUND(current_stock_units - (sold_pre + sold_promo + sold_post)) AS terminal_stock_units,
  ROUND(
    sold_pre * (retail_price - cost_price) +
    (sold_promo + sold_post) * (retail_price * (1 - depth_pct / 100) - cost_price) +
    (current_stock_units - (sold_pre + sold_promo + sold_post)) * (retail_price * 0.28 - cost_price)
  ) AS margin_recovered,
  ROUND(
    (sold_promo + sold_post) * (retail_price - retail_price * (1 - depth_pct / 100))
  ) AS margin_given_away
FROM volumes
ORDER BY start_week, depth_pct;
